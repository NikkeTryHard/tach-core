```rust
/// Result from executing a single hook function
#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct HookResult {
    /// Return value from hook (JSON-serialized for flexibility)
    pub return_value: Option<String>,
    /// All return values when using AllResults aggregation
    pub all_values: Vec<String>,
    /// Side effects captured during hook execution
    pub effects: Vec<HookEffect>,
    /// Source file of the hook that produced this result
    pub source: Option<PathBuf>,
    /// Error message if hook failed
    pub error: Option<String>,
}

impl HookResult {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn new_with_value(value: Option<String>) -> Self {
        Self {
            return_value: value,
            ..Default::default()
        }
    }

    pub fn new_with_error(error: String, source: PathBuf) -> Self {
        Self {
            error: Some(error),
            source: Some(source),
            ..Default::default()
        }
    }

    pub fn add_effect(&mut self, effect: HookEffect) {
        self.effects.push(effect);
    }
}

/// How to aggregate results from multiple hook implementations
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum AggregationStrategy {
    /// Return first non-None result (pytest default for most hooks)
    FirstResult,
    /// Collect all results into a list
    AllResults,
    /// No return value expected (side-effect only hooks)
    NoReturn,
}

/// Aggregate multiple hook results based on strategy
pub fn aggregate_results(results: &[HookResult], strategy: AggregationStrategy) -> HookResult {
    let mut aggregated = HookResult::new();

    // Collect all effects from all hooks
    for result in results {
        aggregated.effects.extend(result.effects.clone());

        // Collect errors
        if let Some(ref err) = result.error {
            if aggregated.error.is_none() {
                aggregated.error = Some(err.clone());
                aggregated.source = result.source.clone();
            }
        }
    }

    match strategy {
        AggregationStrategy::FirstResult => {
            // Return first non-None value
            for result in results {
                if result.return_value.is_some() {
                    aggregated.return_value = result.return_value.clone();
                    break;
                }
            }
        }
        AggregationStrategy::AllResults => {
            // Collect all values
            for result in results {
                if let Some(ref val) = result.return_value {
                    aggregated.all_values.push(val.clone());
                }
            }
        }
        AggregationStrategy::NoReturn => {
            // No return value processing needed
        }
    }

    aggregated
}
```

```python
def call_hook_impl(
    conftest_path: str,
    hook_name: str,
    args: dict,
) -> dict:
    """
    Load a conftest.py module and call the specified hook function.

    Args:
        conftest_path: Absolute path to conftest.py file
        hook_name: Name of hook function to call (e.g., "pytest_configure")
        args: Dictionary of arguments to pass to the hook

    Returns:
        dict with keys:
        - return_value: The hook's return value (JSON-serializable)
        - effects: List of effect dicts (SetEnv, ModifySysPath, etc.)
        - error: Error message if hook failed, None otherwise
    """
    import importlib.util
    import json

    result = {
        "return_value": None,
        "effects": [],
        "error": None,
    }

    # Capture state BEFORE
    env_before = dict(os.environ)
    sys_path_before = list(sys.path)

    try:
        # Load the conftest module dynamically
        spec = importlib.util.spec_from_file_location("conftest", conftest_path)
        if spec is None or spec.loader is None:
            result["error"] = f"Could not load module spec from {conftest_path}"
            return result

        module = importlib.util.module_from_spec(spec)

        # Add the conftest's directory to sys.path temporarily for relative imports
        conftest_dir = os.path.dirname(os.path.abspath(conftest_path))
        if conftest_dir not in sys.path:
            sys.path.insert(0, conftest_dir)

        try:
            spec.loader.exec_module(module)
        except Exception as e:
            result["error"] = f"Failed to execute module {conftest_path}: {e}"
            return result

        # Get the hook function
        if not hasattr(module, hook_name):
            result["error"] = f"Function '{hook_name}' not found in {conftest_path}"
            return result

        hook_func = getattr(module, hook_name)

        # Call the hook with provided arguments
        # Filter args to only those the function accepts
        import inspect
        sig = inspect.signature(hook_func)
        valid_args = {k: v for k, v in args.items() if k in sig.parameters}

        return_value = hook_func(**valid_args)

        # Serialize return value if possible
        if return_value is not None:
            try:
                result["return_value"] = json.dumps(return_value) if not isinstance(return_value, str) else return_value
            except (TypeError, ValueError):
                result["return_value"] = str(return_value)

    except Exception as e:
        result["error"] = f"Hook execution failed: {e}"
        return result

    # Capture state AFTER and compute effects
    env_after = dict(os.environ)
    sys_path_after = list(sys.path)

    # Compute environment variable effects
    result["effects"].extend(_compute_env_delta(env_before, env_after))

    # Compute sys.path effects
    result["effects"].extend(_compute_sys_path_delta(sys_path_before, sys_path_after))

    return result
```

```rust
//! Hook Caller - Rust-side orchestration of pytest hook execution
//!
//! This module provides explicit hook invocation from Rust, loading conftest.py
//! modules via PyO3 and calling hook functions by name.

use crate::hooks::{
    aggregate_results, AggregationStrategy, Hook, HookEffect, HookRegistry, HookResult, HookSpec,
};
use anyhow::{Context, Result};
use pyo3::prelude::*;
use pyo3::types::{PyDict, PyList};
use std::collections::HashMap;
use std::path::{Path, PathBuf};

/// Orchestrates hook execution from Rust
pub struct HookCaller {
    registry: HookRegistry,
    /// Cache of loaded conftest modules to avoid reloading
    loaded_modules: HashMap<PathBuf, bool>,
}

impl HookCaller {
    /// Create a new HookCaller with the given registry
    pub fn new(registry: HookRegistry) -> Result<Self> {
        Ok(Self {
            registry,
            loaded_modules: HashMap::new(),
        })
    }

    /// Get hooks for a given hook name, ordered for execution (root -> leaf)
    pub fn get_hooks_for_execution(&self, hook_name: &str) -> Vec<&Hook> {
        let mut hooks: Vec<_> = self
            .registry
            .get_hooks(hook_name)
            .into_iter()
            .collect();

        // Sort by path depth (root conftest first)
        hooks.sort_by(|a, b| {
            let depth_a = a.source.components().count();
            let depth_b = b.source.components().count();
            depth_a.cmp(&depth_b)
        });

        hooks
    }

    /// Execute all registered implementations of a hook
    pub fn call_hook(
        &mut self,
        hook_name: &str,
        args: HashMap<String, String>,
    ) -> Result<HookResult> {
        let hooks = self.get_hooks_for_execution(hook_name);
        if hooks.is_empty() {
            return Ok(HookResult::new());
        }

        let mut results = Vec::new();

        for hook in hooks {
            let result = self.call_single_hook(hook, &args)?;
            results.push(result);
        }

        // Get aggregation strategy from hook spec
        let strategy = self.get_aggregation_strategy(hook_name);
        Ok(aggregate_results(&results, strategy))
    }

    /// Call a single hook function
    fn call_single_hook(
        &mut self,
        hook: &Hook,
        args: &HashMap<String, String>,
    ) -> Result<HookResult> {
        let source_path = hook.source.to_string_lossy().to_string();
        let hook_name = hook.function_name.clone();

        Python::with_gil(|py| {
            self.call_hook_via_python(py, &source_path, &hook_name, args)
        })
        .with_context(|| {
            format!(
                "Failed to call hook '{}' from '{}'",
                hook_name, source_path
            )
        })
    }

    /// Call hook via Python's call_hook_impl function
    fn call_hook_via_python(
        &self,
        py: Python<'_>,
        conftest_path: &str,
        hook_name: &str,
        args: &HashMap<String, String>,
    ) -> Result<HookResult> {
        // Import tach_harness
        let harness = py
            .import("tach_harness")
            .context("Failed to import tach_harness")?;

        // Convert args to Python dict
        let py_args = PyDict::new(py);
        for (key, value) in args {
            py_args.set_item(key, value)?;
        }

        // Call call_hook_impl
        let result = harness
            .getattr("call_hook_impl")?
            .call1((conftest_path, hook_name, py_args))?;

        // Extract result dict
        let result_dict = result.downcast::<PyDict>()?;

        let mut hook_result = HookResult::new();

        // Extract return_value
        if let Ok(Some(val)) = result_dict.get_item("return_value") {
            if !val.is_none() {
                hook_result.return_value = val.extract().ok();
            }
        }

        // Extract error
        if let Ok(Some(err)) = result_dict.get_item("error") {
            if !err.is_none() {
                hook_result.error = err.extract().ok();
                hook_result.source = Some(PathBuf::from(conftest_path));
            }
        }

        // Extract effects
        if let Ok(Some(effects_list)) = result_dict.get_item("effects") {
            if let Ok(effects) = effects_list.downcast::<PyList>() {
                hook_result.effects = convert_py_effects_to_hook_effects(effects);
            }
        }

        Ok(hook_result)
    }

    /// Get the aggregation strategy for a hook
    fn get_aggregation_strategy(&self, hook_name: &str) -> AggregationStrategy {
        match hook_name {
            // Hooks that return first non-None result
            "pytest_collection_modifyitems" => AggregationStrategy::NoReturn,
            "pytest_runtest_setup" => AggregationStrategy::NoReturn,
            "pytest_runtest_teardown" => AggregationStrategy::NoReturn,
            "pytest_configure" => AggregationStrategy::NoReturn,
            "pytest_sessionstart" => AggregationStrategy::NoReturn,
            "pytest_sessionfinish" => AggregationStrategy::NoReturn,
            // Hooks that aggregate results
            "pytest_runtest_makereport" => AggregationStrategy::FirstResult,
            // Default
            _ => AggregationStrategy::FirstResult,
        }
    }
}

/// Convert Python effect dicts to Rust HookEffect
fn convert_py_effects_to_hook_effects(py_list: &Bound<'_, PyList>) -> Vec<HookEffect> {
    let mut effects = Vec::new();

    for item in py_list.iter() {
        let Ok(dict) = item.downcast::<PyDict>() else {
            continue;
        };

        let effect_type: String = match dict.get_item("type") {
            Ok(Some(t)) => match t.extract() {
                Ok(s) => s,
                Err(_) => continue,
            },
            _ => continue,
        };

        let effect = match effect_type.as_str() {
            "SetEnv" => {
                let key: String = dict
                    .get_item("key")
                    .ok()
                    .flatten()
                    .and_then(|v| v.extract().ok())
                    .unwrap_or_default();
                let value: String = dict
                    .get_item("value")
                    .ok()
                    .flatten()
                    .and_then(|v| v.extract().ok())
                    .unwrap_or_default();
                HookEffect::SetEnv { key, value }
            }
            "ModifySysPath" => {
                let action_str: String = dict
                    .get_item("action")
                    .ok()
                    .flatten()
                    .and_then(|v| v.extract().ok())
                    .unwrap_or_else(|| "Append".to_string());
                let path: String = dict
                    .get_item("path")
                    .ok()
                    .flatten()
                    .and_then(|v| v.extract().ok())
                    .unwrap_or_default();

                let action = match action_str.as_str() {
                    "Prepend" => crate::hooks::SysPathAction::Prepend,
                    "Remove" => crate::hooks::SysPathAction::Remove,
                    _ => crate::hooks::SysPathAction::Append,
                };

                HookEffect::ModifySysPath { action, path }
            }
            _ => HookEffect::NoEffect,
        };

        effects.push(effect);
    }

    effects
}
```

```python
def call_collection_modifyitems(
    conftest_path: str,
    items: list[str],
) -> dict:
    """
    Call pytest_collection_modifyitems hook and track item changes.

    Args:
        conftest_path: Path to conftest.py containing the hook
        items: List of test node IDs to pass to the hook

    Returns:
        dict with keys:
        - reordered: True if order changed
        - removed: List of removed item IDs
        - new_order: List of items after modification
        - error: Error message if failed
    """
    import importlib.util

    result = {
        "reordered": False,
        "removed": [],
        "new_order": list(items),
        "error": None,
    }

    original_items = list(items)
    mutable_items = list(items)

    try:
        # Load conftest module
        spec = importlib.util.spec_from_file_location("conftest", conftest_path)
        if spec is None or spec.loader is None:
            result["error"] = f"Could not load {conftest_path}"
            return result

        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)

        if not hasattr(module, "pytest_collection_modifyitems"):
            result["error"] = "pytest_collection_modifyitems not found"
            return result

        hook_func = getattr(module, "pytest_collection_modifyitems")

        # Call hook - it modifies items in place
        hook_func(items=mutable_items)

        result["new_order"] = mutable_items

        # Detect removals
        removed = [item for item in original_items if item not in mutable_items]
        result["removed"] = removed

        # Detect reordering (items that are in both but different positions)
        common_items = [item for item in original_items if item in mutable_items]
        new_positions = [mutable_items.index(item) for item in common_items if item in mutable_items]
        result["reordered"] = new_positions != sorted(new_positions)

    except Exception as e:
        result["error"] = str(e)

    return result
```

```rust
//! Hook Dependency Graph
//!
//! Manages hook execution order based on conftest.py hierarchy
//! and wrapper specifications.

use crate::hooks::{Hook, HookSpec};
use std::collections::HashMap;
use std::path::PathBuf;

/// Manages hook dependencies and execution order
#[derive(Debug, Default)]
pub struct HookDependencyGraph {
    /// Hooks indexed by name
    hooks: HashMap<String, Vec<Hook>>,
    /// Wrapper hooks (hookwrapper=True)
    wrappers: HashMap<String, Vec<Hook>>,
}

impl HookDependencyGraph {
    pub fn new() -> Self {
        Self::default()
    }

    /// Add a hook to the graph
    pub fn add_hook(&mut self, hook: Hook) {
        let name = hook.spec.name.clone();

        if hook.is_wrapper {
            self.wrappers.entry(name).or_default().push(hook);
        } else {
            self.hooks.entry(name).or_default().push(hook);
        }
    }

    /// Get hooks in execution order (root conftest -> leaf conftest)
    pub fn get_execution_order(&self, hook_name: &str) -> Vec<&Hook> {
        let mut hooks: Vec<_> = self
            .hooks
            .get(hook_name)
            .map(|v| v.iter().collect())
            .unwrap_or_default();

        // Sort by path depth (fewer components = closer to root)
        hooks.sort_by(|a, b| {
            let depth_a = a.source.components().count();
            let depth_b = b.source.components().count();
            depth_a.cmp(&depth_b)
        });

        hooks
    }

    /// Get wrapper hooks for a given hook name
    pub fn get_wrappers(&self, hook_name: &str) -> Vec<&Hook> {
        self.wrappers
            .get(hook_name)
            .map(|v| v.iter().collect())
            .unwrap_or_default()
    }

    /// Get all hook names registered
    pub fn hook_names(&self) -> Vec<&String> {
        let mut names: Vec<_> = self.hooks.keys().collect();
        for name in self.wrappers.keys() {
            if !names.contains(&name) {
                names.push(name);
            }
        }
        names
    }
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PluginStatus {
    /// Plugin is fully supported
    Supported {
        description: String,
    },
    /// Plugin is partially supported with limitations
    Partial {
        description: String,
        limitations: Vec<String>,
    },
    /// Plugin functionality is superseded by Tach
    Superseded {
        description: String,
        tach_equivalent: String,
    },
    /// Plugin is known to be incompatible
    Incompatible {
        reason: String,
    },
    /// Plugin is unknown (may or may not work)
    Unknown,
}

/// Registry of pytest plugins and their compatibility
pub struct PluginRegistry {
    plugins: HashMap<String, PluginStatus>,
    disabled: HashSet<String>,
    priority: Vec<String>,
}

impl PluginRegistry {
    pub fn new() -> Self {
        let mut registry = Self {
            plugins: HashMap::new(),
            disabled: HashSet::new(),
            priority: Vec::new(),
        };

        registry.register_known_plugins();
        registry
    }

    fn register_known_plugins(&mut self) {
        // Supported plugins
        self.plugins.insert(
            "pytest-django".to_string(),
            PluginStatus::Supported {
                description: "Django test support via marker detection".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-asyncio".to_string(),
            PluginStatus::Supported {
                description: "Async test support via is_async detection".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-mock".to_string(),
            PluginStatus::Supported {
                description: "Mocking fixtures work normally".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-env".to_string(),
            PluginStatus::Supported {
                description: "Environment variables captured via effect recording".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-randomly".to_string(),
            PluginStatus::Supported {
                description: "Test randomization works normally".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-timeout".to_string(),
            PluginStatus::Partial {
                description: "Timeout support".to_string(),
                limitations: vec![
                    "Use Tach's native --timeout flag for better integration".to_string(),
                ],
            },
        );

        // Superseded plugins
        self.plugins.insert(
            "pytest-xdist".to_string(),
            PluginStatus::Superseded {
                description: "Parallel test execution".to_string(),
                tach_equivalent: "Tach's native -n flag with worker pool".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-forked".to_string(),
            PluginStatus::Superseded {
                description: "Forked test execution".to_string(),
                tach_equivalent: "Tach's zygote/fork model".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-parallel".to_string(),
            PluginStatus::Superseded {
                description: "Parallel test execution".to_string(),
                tach_equivalent: "Tach's native parallelism".to_string(),
            },
        );
        self.plugins.insert(
            "pytest-cov".to_string(),
            PluginStatus::Superseded {
                description: "Coverage collection".to_string(),
                tach_equivalent: "Tach's --coverage flag with PEP 669".to_string(),
            },
        );

        // Incompatible plugins
        self.plugins.insert(
            "pytest-sugar".to_string(),
            PluginStatus::Incompatible {
                reason: "Terminal manipulation conflicts with Tach's progress display".to_string(),
            },
        );
    }

    /// Get the compatibility status of a plugin
    pub fn get_plugin_status(&self, plugin_name: &str) -> PluginStatus {
        self.plugins
            .get(plugin_name)
            .cloned()
            .unwrap_or(PluginStatus::Unknown)
    }

    /// Disable a plugin by name
    pub fn disable_plugin(&mut self, plugin_name: &str) {
        self.disabled.insert(plugin_name.to_string());
    }

    /// Enable a previously disabled plugin
    pub fn enable_plugin(&mut self, plugin_name: &str) {
        self.disabled.remove(plugin_name);
    }

    /// Check if a plugin is disabled
    pub fn is_disabled(&self, plugin_name: &str) -> bool {
        self.disabled.contains(plugin_name)
    }

    /// Set plugin execution priority (first = highest)
    pub fn set_priority(&mut self, order: Vec<String>) {
        self.priority = order;
    }

    /// Get all known plugin names
    pub fn known_plugins(&self) -> Vec<&String> {
        self.plugins.keys().collect()
    }
}

impl Default for PluginRegistry {
    fn default() -> Self {
        Self::new()
    }
}
```

```python
"""End-to-end integration tests for hook system."""
import os

@pytest.fixture

    """Create a test project with conftest hooks."""
    # Root conftest
    (tmp_path / "conftest.py").write_text("""
import os
def pytest_configure(config):
    os.environ["ROOT_CONFIGURED"] = "yes"

def pytest_collection_modifyitems(items):
    # Add marker to all items
    pass
""")

    # Tests directory
    tests_dir = tmp_path / "tests"
    tests_dir.mkdir()

    (tests_dir / "conftest.py").write_text("""
import os
def pytest_configure(config):
    os.environ["TESTS_CONFIGURED"] = "yes"
""")

    (tests_dir / "test_example.py").write_text("""
import os

def test_hooks_ran():
    assert os.environ.get("ROOT_CONFIGURED") == "yes"
    assert os.environ.get("TESTS_CONFIGURED") == "yes"
""")

    return tmp_path


def test_hooks_execute_in_order(project_with_hooks):
    """Hooks should execute from root conftest to leaf conftest."""
    result = subprocess.run(
        ["tach", str(project_with_hooks / "tests")],
        capture_output=True,
        text=True,
        cwd=project_with_hooks,
    )

    # Test should pass if hooks ran in correct order
    assert result.returncode == 0, f"stdout: {result.stdout}\nstderr: {result.stderr}"
```
